import SwiftNetworkKit
import Foundation

// 1. Type-Safe Endpoint Definition
public struct GitHubUserEndpoint: Endpoint {
    public typealias Response = GitHubUser
    
    public var baseURL: URL {
        URL(string: "https://api.github.com")!
    }
    
    public var path: String {
        "/users/octocat"
    }
    
    public var method: HTTPMethod {
        .get
    }
    
    public var authRequirement: AuthRequirement {
        .none
    }

    public var headers: [String: String] {
        [
            "Accept": "application/vnd.github.v3+json"
        ]
    }
}

// --- Decodable Response Model ---
public struct GitHubUser: Codable, Sendable {
    public let id: Int
    public let name: String
    public let status: String
    public let createdAt: Date?
}

// --- Client Configuration ---
import SwiftNetworkKit

// 2. Initialize NetworkClient
var config = NetworkConfiguration(
    baseURL: URL(string: "https://api.github.com")!,
    headers: ["User-Agent": "MyiOSApp/1.0"]
)

// No Authorization header required (.none)
config.retryPolicy = ExponentialBackoffPolicy(maxAttempts: 3, fullJitter: true)
// SPKI SHA-256 Public Key Pinning (Renewal-safe)
config.sslPinning = .spkiSHA256(["sha256/9kE7yZ6W+V2r0x7G3h5...="])
config.logLevel = .verbose

let client = NetworkClient(configuration: config)

// --- Call Execution ---
// 3. Modern Swift 6 Async/Await Execution
do {
    let endpoint = GitHubUserEndpoint()
    let result: GitHubUser = try await client.request(endpoint)
    print("Received typed response: \(result)")
} catch let error as NetworkError {
    switch error {
    case .unauthorized:
        print("Session expired or token invalid")
    case .sslPinningFailed(let host):
        print("Security warning: SSL pin mismatch on \(host)")
    case .httpError(let statusCode, _):
        print("HTTP Error \(statusCode)")
    default:
        print("Network request failed: \(error.localizedDescription)")
    }
}

SwiftNetworkKit vs. Alamofire

Modern Swift 6 Strict Concurrency vs. Legacy Callback Architecture

SwiftNetworkKit Advantages

  • Pure Swift 6 Actor-isolated concurrency with 0 data races
  • Zero external dependencies (pure Apple standard frameworks)
  • Single-flight 401 token refresh built into the actor layer
  • Built-in SPKI SHA-256 Public Key Pinning with development discovery mode
  • Automatic Bearer token & password redaction in logs
  • Native AsyncStream reachability monitoring

Alamofire Tradeoffs

  • Historical codebase originally designed around completion handlers & GCD locks
  • Token refresh retriers require manual mutex / semaphore synchronizations
  • Heavier binary footprint with thousands of lines of legacy code
  • Logging does not redact sensitive Bearer tokens out-of-the-box

FULL ARCHITECTURE

Enterprise Features Built For Scale

Engineered with clean architectural separation, zero third-party dependencies, and native Swift 6 concurrency primitives.

Swift 6 Strict Concurrency

Fully annotated with Sendable conformances and thread-safe actor boundaries. Guarantees 0 data races under the Swift 6 compiler.

SPKI Public Key Pinning

Zero third-party security libs. Pins raw Subject Public Key Info SHA-256 hashes or bundled certificates with native Trust evaluation.

Jitter & Rate Limit Engine

Full and equal jitter prevent thundering herds on microservices. Automatically parses and respects HTTP Retry-After headers.

RFC 7578 Multipart Uploads

Streams gigabyte-sized files straight from local disk into the transport socket with byte-level progress reporting (ProgressEvent).

Redacting Privacy Logger

Logs cURL commands and status codes without ever leaking Bearer tokens, passwords, or sensitive API keys in the Xcode debug console.

AsyncStream Reachability

Native PathNetworkMonitor wraps Network framework. Yields statusUpdates() and connectionRestored() AsyncStreams.
